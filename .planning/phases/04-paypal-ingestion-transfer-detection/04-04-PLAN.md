```php
public function reclassify(
    string $newType,
    CurrentUser $currentUser,
    DatabaseManager $db,
): void {
    if (! in_array($newType, Transaction::TYPES, true)) {
        throw new \InvalidArgumentException("Invalid transaction type: '{$newType}'");
    }

    $user = $currentUser->user();

    /** @var Transaction $tx */
    $tx = Transaction::query()
        ->where('id', $this->transactionId)
        ->where('user_id', $user->id)
        ->firstOrFail();

    $partnerId = $tx->pair_transaction_id;
    $oldType = $tx->type;

    $db->connection()->transaction(static function () use ($tx, $newType, $partnerId, $user): void {
        $tx->type = $newType;
        if (! in_array($newType, ['transfer_out', 'transfer_in'], true)) {
            $tx->pair_transaction_id = null;
        }
        $tx->save();

        if ($partnerId !== null && ! in_array($newType, ['transfer_out', 'transfer_in'], true)) {
            Transaction::query()
                ->where('user_id', $user->id)
                ->where('id', $partnerId)
                ->update(['pair_transaction_id' => null]);
        }
    });

    // Toast on success (Filament-notifications-style or Flux notification).
    // Use the project's existing notification convention — read Triage.php or
    // similar Livewire components for the canonical dispatch pattern.
    $this->dispatch('toast', [
        'message' => $partnerId !== null && ! in_array($newType, ['transfer_out', 'transfer_in'], true)
            ? "Reclassified to {$newType} — pair removed"
            : "Reclassified to {$newType}",
    ]);
}
```

```php
<?php

declare(strict_types=1);

namespace Modules\Chains\Internal\Resolvers;

use Illuminate\Database\DatabaseManager;
use Modules\Chains\Models\ChainLink;
use Modules\Core\Models\User;
use Modules\Core\Public\Contracts\Clock;
use Modules\Ledger\Models\Transaction;
use Modules\Ledger\Public\Services\FingerprintComposer;
use Modules\Transfers\Public\Services\PairLookup;

/**
 * PayPal funding-chain resolver (CHN-01 + CHN-02 + D-106).
 *
 * Two-arm algorithm:
 *
 *   1. Deterministic arm (CHN-01 + D-106 close-out) — inspect
 *      rawPayload.events[] for funding-source IBAN hints. When a
 *      "General Withdrawal" / "Bankstorting" / "Transfer to bank"
 *      event memo carries an IBAN matching one of the user's own
 *      accounts AND an equal-and-opposite transfer_in exists on that
 *      account within ±3 days: create a chain_link of confidence=1.0,
 *      state='confirmed', resolver='auto'.
 *
 *   2. Fuzzy arm (CHN-02) — when arm 1 misses, score candidate
 *      transfer_in rows by Levenshtein-normalised-merchant similarity
 *      + amount band (±2%) + date window (±3 days). Best score >=0.6
 *      surfaces as state='candidate', resolver='auto'.
 *
 * Resolver writes chain_links ONLY (D-84 invariant) — never mutates
 * transactions. BoundaryArchTest enforces.
 *
 * Cross-user safety: every query filters on $user->id first (FND-03).
 *
 * Raw DatabaseManager query builder per phpstan-strict-rules
 * staticMethod.dynamicCall (PairTransferCandidates is the canonical
 * shape).
 */
final class PaypalFundingResolver
{
    public const AMOUNT_BAND_PERCENT = 2;
    public const DATE_WINDOW_DAYS = 3;
    public const FUZZY_MIN_CONFIDENCE = 0.6;

    /** @var list<string> Events whose memo D-106 inspects for an IBAN. */
    private const FUNDING_EVENT_TYPES = ['General Withdrawal', 'Bankstorting', 'Transfer to bank'];

    public function __construct(
        private readonly DatabaseManager $db,
        private readonly Clock $clock,
        private readonly FingerprintComposer $fingerprints,
        private readonly PairLookup $pairLookup,
    ) {}

    public function resolveForUser(User $user): void
    {
        $connection = $this->db->connection();

        // SELECT every PayPal-account transaction that:
        //   - is type='expense' or 'transfer_out' (the chain-from side)
        //   - belongs to $user
        //   - lacks any chain_link of kind='paypal_funding' from itself
        //     (any state — confirmed OR candidate OR rejected; rejected
        //      pairs use the per-pair pre-insert guard below to stay
        //      non-re-proposed)
        $candidateRows = $connection
            ->table('transactions')
            ->join('accounts', 'accounts.id', '=', 'transactions.account_id')
            ->leftJoin('chain_links', function ($join) {
                $join->on('chain_links.from_transaction_id', '=', 'transactions.id')
                     ->where('chain_links.kind', 'paypal_funding');
            })
            ->where('transactions.user_id', $user->id)
            ->where('accounts.kind', 'paypal')
            ->whereIn('transactions.type', ['expense', 'transfer_out'])
            ->whereNull('chain_links.id')
            ->select(
                'transactions.id',
                'transactions.counterparty_name',
                'transactions.counterparty_normalized',
                'transactions.amount_minor',
                'transactions.settled_amount_minor',
                'transactions.currency',
                'transactions.settled_currency',
                'transactions.posted_at',
                'transactions.booked_at',
                'transactions.raw_payload',
                'transactions.type',
            )
            ->get();

        foreach ($candidateRows as $row) {
            // Hydrate the Transaction model just for raw_payload + Carbon casts.
            /** @var ?Transaction $tx */
            $tx = Transaction::query()->where('id', $row->id)->where('user_id', $user->id)->first();
            if ($tx === null) {
                continue;
            }

            // Arm 1: deterministic (D-106 close-out)
            $link = $this->deterministicMatch($tx, $user);
            if ($link !== null) {
                $this->insertChainLinkIfNotExists($link, $user);
                continue;
            }

            // Arm 2: fuzzy (CHN-02)
            $link = $this->fuzzyMatch($tx, $user);
            if ($link !== null) {
                $this->insertChainLinkIfNotExists($link, $user);
            }
        }
    }

    /**
     * @return ?array<string, mixed> chain_link attributes ready to insert, or null
     */
    private function deterministicMatch(Transaction $tx, User $user): ?array
    {
        $payload = $tx->raw_payload;
        if (! is_array($payload) && ! ($payload instanceof \ArrayAccess)) {
            return null;
        }

        $events = $payload['events'] ?? [];
        if (! is_array($events) || $events === []) {
            return null;
        }

        foreach ($events as $event) {
            $eventType = $event['type'] ?? '';
            if (! is_string($eventType) || ! in_array($eventType, self::FUNDING_EVENT_TYPES, true)) {
                continue;
            }
            $eventRow = $event['row'] ?? [];
            $iban = $this->extractIbanFromEventRow(is_array($eventRow) ? $eventRow : []);
            if ($iban === null) {
                continue;
            }
            // Find user's account with that IBAN.
            $account = $this->db->connection()->table('accounts')
                ->where('user_id', $user->id)
                ->where('iban', $iban)
                ->first(['id']);
            if ($account === null) {
                continue;
            }
            $accountId = self::toInt($account->id);

            // Find the partner transfer_in on that account within ±N days,
            // equal-and-opposite amount.
            $partnerId = $this->findPartnerOnAccount(
                $accountId,
                -self::toInt($tx->amount_minor),
                $tx->booked_at,
                $user,
            );
            if ($partnerId === null) {
                continue;
            }

            $normalisedMerchant = self::toString($tx->counterparty_normalized);
            $signatureHash = hash('sha256', $normalisedMerchant.'|'.$iban);

            return [
                'user_id' => $user->id,
                'from_transaction_id' => self::toInt($tx->id),
                'to_transaction_id' => $partnerId,
                'kind' => 'paypal_funding',
                'state' => 'confirmed',
                'confidence' => 1.0,
                'resolver' => 'auto',
                'evidence' => [
                    'matched_reference_id' => self::toStringOrNull($eventRow['Reference Txn ID'] ?? null),
                    'matched_iban' => $iban,
                    'event_type' => $eventType,
                    'signature_hash' => $signatureHash,
                ],
            ];
        }
        return null;
    }

    /**
     * @return ?array<string, mixed>
     */
    private function fuzzyMatch(Transaction $tx, User $user): ?array
    {
        $normalisedMerchant = self::toString($tx->counterparty_normalized);
        $settledMinor = abs(self::toInt($tx->settled_amount_minor));
        if ($settledMinor === 0) {
            return null;
        }
        $amountBand = (int) round($settledMinor * (self::AMOUNT_BAND_PERCENT / 100));

        $candidates = $this->db->connection()->table('transactions')
            ->where('user_id', $user->id)
            ->where('id', '<>', self::toInt($tx->id))
            ->where('type', 'transfer_in')
            ->whereBetween('settled_amount_minor', [
                -($settledMinor + $amountBand),
                -($settledMinor - $amountBand),
            ])
            ->whereBetween('posted_at', [
                $tx->posted_at->copy()->subDays(self::DATE_WINDOW_DAYS)->toDateTimeString(),
                $tx->posted_at->copy()->addDays(self::DATE_WINDOW_DAYS)->toDateTimeString(),
            ])
            ->limit(20)
            ->get(['id', 'counterparty_normalized', 'posted_at', 'settled_amount_minor', 'account_id']);

        $bestId = null;
        $bestScore = 0.0;
        $bestRow = null;
        foreach ($candidates as $candidate) {
            $merchantSim = $this->levenshteinSimilarity(
                $normalisedMerchant,
                self::toString($candidate->counterparty_normalized),
            );
            $amountDelta = abs($settledMinor - abs(self::toInt($candidate->settled_amount_minor)));
            $amountSim = max(0.0, 1.0 - ($amountDelta / max(1, $settledMinor)));
            $dateDelta = abs(\Carbon\CarbonImmutable::parse(self::toString($candidate->posted_at))
                ->diffInDays($tx->posted_at, false));
            $dateSim = max(0.0, 1.0 - ($dateDelta / self::DATE_WINDOW_DAYS));

            // Weighted score per RESEARCH Example 2 (discretion-locked weights).
            $score = (0.5 * $merchantSim) + (0.3 * $amountSim) + (0.2 * $dateSim);

            if ($score > $bestScore && $score >= self::FUZZY_MIN_CONFIDENCE) {
                $bestScore = $score;
                $bestId = self::toInt($candidate->id);
                $bestRow = $candidate;
            }
        }

        if ($bestId === null || $bestRow === null) {
            return null;
        }

        $accountIban = $this->ibanForAccountId(self::toInt($bestRow->account_id), $user) ?? '';
        $signatureHash = hash('sha256', $normalisedMerchant.'|'.$accountIban);

        return [
            'user_id' => $user->id,
            'from_transaction_id' => self::toInt($tx->id),
            'to_transaction_id' => $bestId,
            'kind' => 'paypal_funding',
            'state' => 'candidate',
            'confidence' => round($bestScore, 3),
            'resolver' => 'auto',
            'evidence' => [
                'merchant_similarity' => round($merchantSim, 3),
                'amount_delta_minor' => $amountDelta,
                'date_delta_days' => (int) $dateDelta,
                'signature_hash' => $signatureHash,
            ],
        ];
    }

    /**
     * Per-pair pre-insert guard — if ANY chain_link (any state, including
     * rejected) already exists for this (from, to, kind, user_id) tuple,
     * skip the insert. This is how rejected pairs stay non-re-proposed
     * across resolver passes (D-89 + the rejected-pair stay-rejected
     * behavior).
     *
     * @param array<string, mixed> $link
     */
    private function insertChainLinkIfNotExists(array $link, User $user): void
    {
        $exists = $this->db->connection()->table('chain_links')
            ->where('user_id', $user->id)
            ->where('from_transaction_id', $link['from_transaction_id'])
            ->where('to_transaction_id', $link['to_transaction_id'])
            ->where('kind', $link['kind'])
            ->exists();
        if ($exists) {
            return;

        $now = $this->clock->now()->toDateTimeString();
        $this->db->connection()->table('chain_links')->insert([
            'user_id' => $link['user_id'],
            'from_transaction_id' => $link['from_transaction_id'],
            'to_transaction_id' => $link['to_transaction_id'],
            'kind' => $link['kind'],
            'state' => $link['state'],
            'confidence' => $link['confidence'],
            'resolver' => $link['resolver'],

            'updated_at' => $now,

    private function findPartnerOnAccount(int $accountId, int $expectedAmountMinor, \Carbon\CarbonInterface $center, User $user): ?int
    {
        $row = $this->db->connection()->table('transactions')
            ->where('user_id', $user->id)
            ->where('account_id', $accountId)
            ->where('type', 'transfer_in')
            ->where('amount_minor', $expectedAmountMinor)
            ->whereBetween('booked_at', [
                $center->copy()->subDays(self::DATE_WINDOW_DAYS)->toDateTimeString(),
                $center->copy()->addDays(self::DATE_WINDOW_DAYS)->toDateTimeString(),
            ])
            ->orderByRaw('ABS(julianday(booked_at) - julianday(?))', [$center->toDateTimeString()])
            ->first(['id']);
        return $row === null ? null : self::toInt($row->id);
    }

    /**
     * Extract an IBAN from an event row's free-text memo or dedicated field.
     * The PayPal NL CSV's General Withdrawal row carries the destination
     * IBAN in the "Naam"/"Memo" column per Phase 4 D-106. Lenient regex —
     * the IBAN format is `[A-Z]{2}\d{2}[A-Z0-9]{1,30}` and the script's
     * synthesised fixture uses NL57ASNB0123456789 as the matching target.
     *
     * @param array<string, mixed> $eventRow
     */
    private function extractIbanFromEventRow(array $eventRow): ?string
    {
        $haystack = '';
        foreach (['Naam', 'Memo', 'Note', 'Description', 'Omschrijving'] as $key) {
            if (isset($eventRow[$key]) && is_string($eventRow[$key])) {
                $haystack .= ' '.$eventRow[$key];
            }
        }
        if ($haystack === '') {
            return null;
        }
        if (preg_match('/\b([A-Z]{2}\d{2}[A-Z0-9]{8,30})\b/', strtoupper($haystack), $matches) === 1) {
            return $matches[1];
        }
        return null;
    }

    private function ibanForAccountId(int $accountId, User $user): ?string
    {
        $row = $this->db->connection()->table('accounts')
            ->where('user_id', $user->id)
            ->where('id', $accountId)
            ->first(['iban']);
        return $row === null ? null : self::toString($row->iban);
    }

    private function levenshteinSimilarity(string $a, string $b): float
    {
        $maxLen = max(mb_strlen($a), mb_strlen($b));
        if ($maxLen === 0) {
            return 1.0;
        }
        $dist = levenshtein($a, $b);
        return 1.0 - ($dist / $maxLen);
    }

    private static function toInt(mixed $v): int
    {
        return is_numeric($v) ? (int) $v : 0;
    }

    private static function toString(mixed $v): string
    {
        return is_string($v) ? $v : '';
    }

    private static function toStringOrNull(mixed $v): ?string
    {
        return is_string($v) && $v !== '' ? $v : null;
    }
}
```

```php
declare(strict_types=1);

namespace Modules\Chains\Public\Services;

use Illuminate\Database\DatabaseManager;
use Modules\Chains\Public\Dto\ChainLinkRow;
use Modules\Chains\Public\Dto\ChainTree;
use Modules\Chains\Public\Dto\ChainTreeNode;
use Modules\Core\Models\User;
use Modules\Ledger\Public\ValueObjects\Money;

final class ChainLinkQuery
{
    private const MAX_DEPTH = 5;
    private const AUTO_PROMOTE_THRESHOLD = 3;     // D-87

    public function __construct(
        private readonly DatabaseManager $db,
    ) {}

    public function forTransaction(int $transactionId, User $user): ChainTree
    {
        $rootRow = $this->db->connection()->table('transactions')
            ->where('id', $transactionId)
            ->where('user_id', $user->id)
            ->first(['id', 'counterparty_name', 'amount_minor', 'currency',
                     'settled_amount_minor', 'settled_currency', 'booked_at', 'account_id']);
        if ($rootRow === null) {
            throw new \Symfony\Component\HttpKernel\Exception\NotFoundHttpException(
                'Transaction not found.'
            );
        }

        // Walk chain_links starting at the root, building the node array
        // top-down. Bounded at MAX_DEPTH levels.
        $nodes = [];
        $rootNode = $this->makeNode($rootRow, null, 'root', 'Confirmed', $user);
        $nodes[] = $rootNode;

        $frontier = [self::toInt($rootRow->id)];
        $visited = [self::toInt($rootRow->id) => true];
        $depth = 0;

        while ($frontier !== [] && $depth < self::MAX_DEPTH) {
            $links = $this->db->connection()->table('chain_links')
                ->where('user_id', $user->id)
                ->whereIn('from_transaction_id', $frontier)
                ->whereIn('state', ['confirmed', 'candidate'])   // rejected filtered out
                ->orderByDesc('confidence')
                ->get();
            $nextFrontier = [];
            foreach ($links as $link) {
                $toId = self::toInt($link->to_transaction_id);
                if (isset($visited[$toId])) {
                    continue;
                }
                $visited[$toId] = true;
                $nextFrontier[] = $toId;
                $toRow = $this->db->connection()->table('transactions')
                    ->where('id', $toId)
                    ->where('user_id', $user->id)
                    ->first(['id', 'counterparty_name', 'amount_minor', 'currency',
                             'settled_amount_minor', 'settled_currency', 'booked_at', 'account_id']);
                if ($toRow === null) {
                    continue;
                }
                $confidenceTier = $this->confidenceTier(
                    self::toString($link->state),
                    self::toString($link->resolver),
                    (float) $link->confidence,
                );
                $nodes[] = $this->makeNode($toRow, self::toInt($link->id),
                    self::toString($link->kind), $confidenceTier, $user);
            }
            $frontier = $nextFrontier;
            $depth++;
        }

        return new ChainTree(
            rootTransactionId: self::toInt($rootRow->id),
            nodes: $nodes,
        );
    }

    public function openCandidateCount(User $user): int
    {
        return self::toInt($this->db->connection()->table('chain_links')
            ->where('user_id', $user->id)
            ->where('state', 'candidate')
            ->count());
    }

    /** @return list<ChainLinkRow> */
    public function candidatesForReview(
        User $user,
        ?int $cursorId = null,
        ?string $cursorConfidence = null,
        int $limit = 26,
    ): array {
        $query = $this->db->connection()->table('chain_links')
            ->where('user_id', $user->id)
            ->where('state', 'candidate')
            ->orderByDesc('confidence')
            ->orderByDesc('updated_at')
            ->limit($limit);

        if ($cursorId !== null && $cursorConfidence !== null) {
            // Cursor-pagination: skip rows we've already seen.
            $query->where(function ($q) use ($cursorConfidence, $cursorId): void {
                $q->where('confidence', '<', $cursorConfidence)
                  ->orWhere(function ($q2) use ($cursorConfidence, $cursorId): void {
                      $q2->where('confidence', '=', $cursorConfidence)
                         ->where('id', '<', $cursorId);
                  });
            });
        }

        $rows = $query->get();
        $result = [];
        foreach ($rows as $row) {
            $result[] = $this->makeChainLinkRow($row, $user);
        }
        return $result;
    }

    private function confidenceTier(string $state, string $resolver, float $confidence): string
    {
        return match (true) {
            $state === 'confirmed' && $resolver === 'auto' && $confidence === 1.0 => 'Deterministic',
            $state === 'confirmed' => 'Confirmed',
            default => 'Candidate',
        };
    }

    private function makeNode(\stdClass $row, ?int $chainLinkId, string $kind, string $tier, User $user): ChainTreeNode
    {
        $accountName = $this->resolveAccountName(self::toInt($row->account_id), $user);
        $amountMinor = self::toInt($row->settled_amount_minor ?? $row->amount_minor);
        $currency = self::toString($row->settled_currency ?? $row->currency);
        return new ChainTreeNode(
            transactionId: self::toInt($row->id),
            chainLinkId: $chainLinkId,
            counterpartyName: self::toString($row->counterparty_name),
            amount: Money::ofMinor($amountMinor, $currency !== '' ? $currency : 'EUR'),
            bookedAt: \Carbon\CarbonImmutable::parse(self::toString($row->booked_at)),
            accountName: $accountName,
            kind: $kind,
            confidenceTier: $tier,
            children: [],   // ICS bulk-settle fan-out populated below
        );
    }

    private function makeChainLinkRow(\stdClass $row, User $user): ChainLinkRow
    {
        // For each candidate, compute confirmsRemaining = max(0, 3 - count(confirmed-with-same-signature)).
        $evidence = json_decode(self::toString($row->evidence), true);
        $signatureHash = is_array($evidence) ? ($evidence['signature_hash'] ?? null) : null;
        $confirmsRemaining = self::AUTO_PROMOTE_THRESHOLD;
        if (is_string($signatureHash)) {
            $confirmedCount = self::toInt($this->db->connection()->table('chain_links')
                ->where('user_id', $user->id)
                ->where('state', 'confirmed')
                ->whereJsonContains('evidence->signature_hash', $signatureHash)
                ->count());
            $confirmsRemaining = max(0, self::AUTO_PROMOTE_THRESHOLD - $confirmedCount);
        }

        $from = $this->db->connection()->table('transactions')
            ->where('id', $row->from_transaction_id)->where('user_id', $user->id)
            ->first(['counterparty_name', 'settled_amount_minor', 'settled_currency', 'posted_at']);
        $to = $this->db->connection()->table('transactions')
            ->where('id', $row->to_transaction_id)->where('user_id', $user->id)
            ->first(['counterparty_name', 'settled_amount_minor', 'settled_currency', 'posted_at']);

        return new ChainLinkRow(
            chainLinkId: self::toInt($row->id),
            kind: self::toString($row->kind),
            state: self::toString($row->state),
            confidence: (float) $row->confidence,
            fromCounterparty: self::toString($from?->counterparty_name ?? ''),
            fromAmount: Money::ofMinor(self::toInt($from?->settled_amount_minor ?? 0),
                self::toString($from?->settled_currency ?? 'EUR')),
            toCounterparty: self::toString($to?->counterparty_name ?? ''),
            toAmount: Money::ofMinor(self::toInt($to?->settled_amount_minor ?? 0),
                self::toString($to?->settled_currency ?? 'EUR')),
            fromPostedAt: \Carbon\CarbonImmutable::parse(self::toString($from?->posted_at ?? '1970-01-01')),
            toPostedAt: \Carbon\CarbonImmutable::parse(self::toString($to?->posted_at ?? '1970-01-01')),
            confirmsRemaining: $confirmsRemaining,
        );
    }

    private function resolveAccountName(int $accountId, User $user): string
    {
        $row = $this->db->connection()->table('accounts')
            ->where('id', $accountId)->where('user_id', $user->id)
            ->first(['name']);
        return $row === null ? '' : self::toString($row->name);
    }

    private static function toInt(mixed $v): int { return is_numeric($v) ? (int) $v : 0; }
    private static function toString(mixed $v): string { return is_string($v) ? $v : ''; }
}
```

```php
namespace Modules\Chains\Public\Actions;

use Illuminate\Database\DatabaseManager;
use Modules\Chains\Models\ChainLink;
use Modules\Core\Models\User;
use Modules\Core\Public\Contracts\Clock;

final class ConfirmChainLink
{
    private const AUTO_PROMOTE_THRESHOLD = 3;  // D-87

    public function __construct(
        private readonly DatabaseManager $db,
        private readonly Clock $clock,
    ) {}

    public function __invoke(int $chainLinkId, User $user): void
    {
        /** @var ChainLink $link */
        $link = ChainLink::query()
            ->where('id', $chainLinkId)
            ->where('user_id', $user->id)
            ->firstOrFail();    // 404 on cross-user

        $this->db->connection()->transaction(function () use ($link, $user): void {
            $link->state = 'confirmed';
            // Preserve original resolver value — don't flip 'auto' to 'user' here;
            // the resolver wrote 'auto', and the user merely confirmed.
            // (UI distinguishes confirmed/auto = 'Deterministic' vs confirmed/user|rule = 'Confirmed'.)
            $link->save();

            $evidence = is_array($link->evidence) ? $link->evidence : [];
            $signatureHash = $evidence['signature_hash'] ?? null;
            if (! is_string($signatureHash)) {
                return;
            }

            $confirmedCount = (int) $this->db->connection()->table('chain_links')
                ->where('user_id', $user->id)
                ->where('state', 'confirmed')
                ->whereJsonContains('evidence->signature_hash', $signatureHash)
                ->count();

            if ($confirmedCount < self::AUTO_PROMOTE_THRESHOLD) {
                return;
            }

            // Auto-promote every other candidate with the same signature.
            $now = $this->clock->now()->toDateTimeString();
            $this->db->connection()->table('chain_links')
                ->where('user_id', $user->id)
                ->where('state', 'candidate')
                ->whereJsonContains('evidence->signature_hash', $signatureHash)
                ->update([
                    'state'      => 'confirmed',
                    'resolver'   => 'rule',          // distinguish auto-promoted from user-confirmed
                    'updated_at' => $now,
                ]);
        });
    }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User-input chainLinkId ←→ ConfirmChainLink / RejectChainLink | Livewire dispatches user-controlled integer IDs into the action; firstOrFail + user_id filter is the only barrier against cross-user mutation. |

| Chain tree walk depth | Unbounded walk could infinite-loop on a cyclic chain_links graph (theoretically impossible because resolver guards) — MAX_DEPTH=5 + visited[] set defends. |

    `Modules/Ingestion/Internal/Adapters/Asn/AsnCamt053HeaderProfile.php`:

    ```php
    <?php

    declare(strict_types=1);

    namespace Modules\Ingestion\Internal\Adapters\Asn;

    /**
     * CAMT.053 (ISO 20022 bank-to-customer statement) profile for ASN Online
     * Bankieren exports. The library `genkgo/camt` handles every CAMT.053
     * sub-version ASN emits (001.02 / 001.03 / 001.04 / 001.08); the FORMAT
     * constant below is the stable identifier the SourceAdapterRegistry +
     * UploadWizard dropdown reference.
     */
    final class AsnCamt053HeaderProfile
    {
        public const FORMAT = 'asn-camt053';

        /** Accepted file extensions for this format. */
        public const FILE_EXTENSIONS = ['xml'];

        /**
         * ISO 20022 CAMT.053 family namespace pattern. The full URI is
         *   urn:iso:std:iso:20022:tech:xsd:camt.053.001.XX
         * with XX being the two-digit sub-version (02, 03, 04, 08). The
         * regex anchors on the family so any unknown future sub-version
         * (e.g. 001.09) passes the sniffer; the parser library will surface
         * a clear error if the sub-version itself is unsupported.
         */
        public const XML_NAMESPACE_REGEX = '#xmlns(?::\w+)?\s*=\s*"urn:iso:std:iso:20022:tech:xsd:camt\.053\.001\.(\d{2})"#';

        public const SOURCE_ENCODING = 'UTF-8';
    }
    ```

    ```php
    private function sniffAsnCamt053(string $path, string $head): SniffResult
    {
        if (preg_match('/\.xml$/i', $path) !== 1) {
            throw new SniffMismatchException(
                "That file doesn't look like an XML file. Drop in the ASN CAMT.053 XML export."
            );
        }

        if (preg_match(AsnCamt053HeaderProfile::XML_NAMESPACE_REGEX, $head) !== 1) {
            throw new SniffMismatchException(
                "This XML file does not declare an ISO 20022 CAMT.053 namespace. If ASN changed their export format, file an issue."
            );
        }

        return new SniffResult(
            format: AsnCamt053HeaderProfile::FORMAT,
            delimiter: '',
            hasHeader: false,
            encoding: AsnCamt053HeaderProfile::SOURCE_ENCODING,
            columnCount: 0,
        );
    }
    ```

    Note: the `match()` arm dispatch with the new `$head` (8 KB head, BOM already stripped by the existing `sniff()` body) is sufficient — leading XML declaration + leading comments fit in 8 KB; the regex does not anchor to start-of-string.

    namespace Modules\Ingestion\Internal\Adapters\Asn;

    use Carbon\CarbonImmutable;
    use DateTimeImmutable;
    use Generator;
    use Genkgo\Camt\Config;
    use Genkgo\Camt\DTO\Entry;
    use Genkgo\Camt\DTO\EntryTransactionDetail;
    use Genkgo\Camt\Reader;
    use Modules\Ingestion\Public\Contracts\AccountResolver;
    use Modules\Ingestion\Public\Contracts\SourceAdapter;
    use Modules\Ingestion\Public\Dto\SourceTransactionDto;
    use Modules\Ingestion\Public\Exceptions\InvalidAmountException;
    use Modules\Ingestion\Public\Services\HeaderSniffer;
    use Throwable;

    /**
     * Streaming-by-row parser for ASN's CAMT.053 (ISO 20022 bank-to-customer
     * statement) XML export. Delegates to genkgo/camt for the file-level
     * unmarshal, then walks every Statement -> Entry -> TransactionDetail and
     * yields one SourceTransactionDto per TxDtls (batch entries are split).
     *
     * EndToEndId is the canonical source_ref. Secondary refs (AcctSvcrRef,
     * InstrId, TxId, MsgId, MandateId) are preserved verbatim inside
     * rawPayload['sepa'] for downstream consumers.
     */
    final class AsnCamt053Adapter implements SourceAdapter
    {
        public function __construct(
            private readonly HeaderSniffer $sniffer,
        ) {}

        public function format(): string
        {
            return AsnCamt053HeaderProfile::FORMAT;
        }

        public function parse(string $localPath, AccountResolver $accounts): Generator
        {
            $this->sniffer->sniff($localPath, AsnCamt053HeaderProfile::FORMAT);

            // Defence-in-depth: explicitly disable external entity loading on this
            // libxml call. genkgo/camt uses XMLReader which does NOT expand entities
            // by default in PHP 8+, but we add the disabled flag anyway because the
            // input is an arbitrary user upload.
            $previousState = libxml_use_internal_errors(true);
            try {
                $reader = new Reader(Config::getDefault());
                try {
                    $message = $reader->readFile($localPath);
                } catch (Throwable $e) {
                    throw new InvalidAmountException(
                        sprintf('Failed to parse CAMT.053 XML: %s', $e->getMessage()),
                        0,
                        $e,
                    );
                }
            } finally {
                libxml_use_internal_errors($previousState);
            }

            $msgId = $message->getGroupHeader()?->getMessageId();
            $index = 0;

            foreach ($message->getRecords() as $statement) {
                $ownIban = $this->extractOwnIban($statement);
                $accounts->resolve($ownIban);

                foreach ($statement->getEntries() as $entry) {
                    $txDtlsList = $entry->getTransactionDetails();

                    if ($txDtlsList === []) {
                        yield $this->buildDto($entry, null, $ownIban, $index++, $msgId);
                        continue;
                    }

                    $isBatch = count($txDtlsList) > 1;
                    foreach ($txDtlsList as $txDtls) {
                        yield $this->buildDto($entry, $txDtls, $ownIban, $index++, $msgId, isBatch: $isBatch);
                    }
                }
            }
        }

        private function buildDto(
            Entry $entry,
            ?EntryTransactionDetail $txDtls,
            string $ownIban,
            int $rowIndex,
            ?string $msgId,
            bool $isBatch = false,
        ): SourceTransactionDto {
            // Amount: per-TxDtls if a batch entry, else entry-level
            $money = $isBatch && $txDtls?->getAmountDetails() !== null
                ? $txDtls->getAmountDetails()
                : $entry->getAmount();

            // moneyphp/money returns the minor units as a numeric string
            $minor = (int) $money->getAmount();
            $currency = $money->getCurrency()->getCode();

            $cdi = $txDtls?->getCreditDebitIndicator() ?? $entry->getCreditDebitIndicator();
            $signed = $cdi === 'DBIT' ? -$minor : $minor;

            $endToEndId = $txDtls?->getReference()?->getEndToEndId();
            $sourceRef = ($endToEndId !== null && $endToEndId !== '' && $endToEndId !== 'NOTPROVIDED')
                ? $endToEndId
                : null;

            [$cpName, $cpIban] = $this->extractCounterparty($txDtls, $cdi);
            $description = $this->extractRemittance($txDtls);

            $booking = $entry->getBookingDate() ?? $entry->getValueDate() ?? new DateTimeImmutable;
            $value = $entry->getValueDate() ?? $booking;

            $rawPayload = $this->serialiseSepaFragment($entry, $txDtls, $msgId);

            return new SourceTransactionDto(
                bookedAt: CarbonImmutable::instance($booking),
                postedAt: CarbonImmutable::instance($booking),
                valueDate: CarbonImmutable::instance($value),
                ownIban: $ownIban,
                counterpartyIban: $cpIban,
                counterpartyName: $cpName,
                currency: $currency,
                amountMinor: $signed,
                sourceRef: $sourceRef,
                description: $description,
                rawPayload: $rawPayload,
                sourceRowIndex: $rowIndex,
            );
        }

        private function extractOwnIban(\Genkgo\Camt\Camt053\DTO\Statement $stmt): string
        {
            $iban = $stmt->getAccount()?->getIdentification()?->getIban() ?? '';
            return is_string($iban) ? $iban : (string) $iban;
        }

        /**
         * @return array{0: ?string, 1: ?string}  [counterparty name, counterparty IBAN]
         */
        private function extractCounterparty(?EntryTransactionDetail $txDtls, ?string $cdi): array
        {

            //     CRDT/DBIT direction (Debtor for CRDT, Creditor for DBIT)
            // ... defensive null-handling throughout
        }

        private function extractRemittance(?EntryTransactionDetail $txDtls): ?string
        {

        /**
         * @return array<string, mixed>
         */
        private function serialiseSepaFragment(Entry $entry, ?EntryTransactionDetail $txDtls, ?string $msgId): array
        {
            return [
                'sepa' => [
                    'msgId' => $msgId,
                    'acctSvcrRef' => $entry->getAccountServicerReference(),
                    'entryRef' => $entry->getReference(),
                    'batchPaymentId' => $entry->getBatchPaymentId(),

| user-uploaded XML → CAMT parser | User-controlled XML enters `genkgo/camt`'s XMLReader. Classic XXE / billion-laughs / DTD-injection territory. |

$path = base_path($this->config->get('community.corpus.bundled_path'));

$parsed = \Symfony\Component\Yaml\Yaml::parseFile($path, \Symfony\Component\Yaml\Yaml::PARSE_EXCEPTION_ON_INVALID_TYPE);
$entries = $parsed['entries'] ?? [];
$result = [];
foreach ($entries as $i => $raw) {
    try {
        if (! is_array($raw) || empty($raw['pattern']) || empty($raw['name']) || empty($raw['contributor'])) {
            $this->logger->warning('Skipped malformed corpus entry.', ['index' => $i, 'reason' => 'missing required field']);
            continue;
        }
        $result[] = new CorpusEntryDto(
            pattern: (string) $raw['pattern'],
            generalizedPattern: isset($raw['generalized_pattern']) ? (string) $raw['generalized_pattern'] : $this->generalizer->generalize((string) $raw['pattern']),
            name: (string) $raw['name'],
            category: isset($raw['category']) ? (string) $raw['category'] : null,
            region: isset($raw['region']) ? (string) $raw['region'] : null,
            contributor: (string) $raw['contributor'],
        );
    } catch (\Throwable $e) {
        $this->logger->warning('Skipped malformed corpus entry.', ['index' => $i, 'exception_class' => $e::class, 'exception_message' => $e->getMessage()]);
    }
}
return $result;

```
final class SeedCommunityCorpus {
    public function __construct(
        private readonly \Modules\Community\Internal\Corpus\CorpusLoader $loader,
        private readonly \Illuminate\Database\DatabaseManager $db,
        private readonly \Psr\Log\LoggerInterface $logger,
    ) {}

    public function handle(\Modules\Core\Public\Events\UserInstalled $event): void {
        $entries = $this->loader->loadBundled();
        $now = now();
        foreach ($entries as $entry) {
            try {
                $this->db->connection()->table('community_merchant_mappings')->updateOrInsert(
                    ['pattern' => $entry->pattern, 'user_id' => null],
                    [
                        'generalized_pattern' => $entry->generalizedPattern,
                        'name' => $entry->name,
                        'category' => $entry->category,
                        'region' => $entry->region,
                        'contributor' => $entry->contributor,
                        'updated_at' => $now,
                    ],
                );
            } catch (\Throwable $e) {
                $this->logger->warning('Skipped corpus entry on seed.', [
                    'pattern' => $entry->pattern,
                    'exception_class' => $e::class,
                    'exception_message' => $e->getMessage(),
                ]);
            }
        }
    }
}
```

final class OpenExternalUrlAction {
    private const ALLOWED_HOSTS = ['github.com'];

    public function __construct(
        private readonly \Native\Desktop\Contracts\Shell $shell,
        private readonly \Psr\Log\LoggerInterface $logger,
    ) {}

    public function __invoke(string $url): void {
        if (! filter_var($url, FILTER_VALIDATE_URL) || ! str_starts_with($url, 'https://')) {
            throw new \InvalidArgumentException('OpenExternalUrlAction: URL must be https://, got: '.$url);
        }
        $host = parse_url($url, PHP_URL_HOST);
        if ($host === false || $host === null || ! in_array($host, self::ALLOWED_HOSTS, true)) {
            throw new \InvalidArgumentException('OpenExternalUrlAction: host not allow-listed, got: '.($host ?: 'null'));
        }
        $this->shell->openExternal($url);
        $this->logger->info('OpenExternalUrlAction: launched system browser.', ['url' => $url]);
    }
}

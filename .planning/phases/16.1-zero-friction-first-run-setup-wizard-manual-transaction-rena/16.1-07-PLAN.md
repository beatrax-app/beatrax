```
final class LongestCommonPrefix {
    public function compute(array $patterns): string {
        if (count($patterns) < 2) {
            throw new \InvalidArgumentException('LCP requires at least 2 patterns; got '.count($patterns));
        }
        $first = mb_strtolower(reset($patterns));
        if ($first === '') return '';
        $minLen = mb_strlen($first);
        foreach ($patterns as $p) {
            $pLower = mb_strtolower($p);
            $minLen = min($minLen, mb_strlen($pLower));
            $i = 0;
            while ($i < $minLen && mb_substr($first, $i, 1) === mb_substr($pLower, $i, 1)) { $i++; }
            $minLen = $i;
            if ($minLen === 0) return '';
        }
        $result = trim(mb_substr($first, 0, $minLen));
        if (mb_strlen($result) < 4) return '';
        return $result;
    }
}
```

```
return $this->db->connection()->transaction(function () use ($user, $aliasIds, $friendlyName, $generalizedPattern): MerchantAlias {
    $rows = MerchantAlias::query()->where('user_id', $user->id)->whereIn('id', $aliasIds)->get();
    if ($rows->count() !== count($aliasIds)) {
        throw new \Symfony\Component\HttpKernel\Exception\NotFoundHttpException('One or more alias IDs not found for this user');
    }
    if ($rows->count() < 2) {
        throw new \InvalidArgumentException('Merge requires at least 2 aliases');
    }
    $surviving = $rows->sortBy('id')->first();
    $absorbed = $rows->filter(fn ($r) => $r->id !== $surviving->id);
    $mergedFrom = $absorbed->map(fn ($r) => ['pattern' => $r->pattern, 'generalized_pattern' => $r->generalized_pattern, 'friendly_name' => $r->friendly_name, 'merged_at' => now()->toIso8601String()])->values()->all();
    $surviving->update(['friendly_name' => $friendlyName, 'generalized_pattern' => $generalizedPattern, 'merged_from' => $mergedFrom]);
    MerchantAlias::query()->whereIn('id', $absorbed->pluck('id')->all())->delete();
    return $surviving->fresh();
```

**Modify `Modules/Import/Internal/Http/Livewire/AliasesSettingsPage.php`** — add file-upload properties + parse/confirm methods:
```
use \Livewire\WithFileUploads;
public ?\Livewire\Features\SupportFileUploads\TemporaryUploadedFile $importFile = null;
public array $importDiff = ['new' => [], 'unchanged' => [], 'conflicts' => []];
/** @var array<string, string> */
public array $conflictResolutions = [];
public ?string $importError = null;

public function rules(): array {
    return [
        'importFile' => ['nullable', 'file', 'extensions:yaml,yml', 'max:1024'],
    ];
}

public function parseUpload(AliasYamlImporter $importer, CurrentUser $currentUser): void {
    $this->validate();
    if ($this->importFile === null) return;
    $this->importError = null;
    try {
        $content = file_get_contents($this->importFile->getRealPath());
        $entries = $importer->parse($content);
        $this->importDiff = $importer->diff($currentUser->user(), $entries);
        // initialize conflictResolutions with 'keep' default for each conflict
        $this->conflictResolutions = [];
        foreach ($this->importDiff['conflicts'] as $entry) {
            $this->conflictResolutions[$entry->pattern] = 'keep';
        }
    } catch (\InvalidArgumentException $e) {
        $this->importError = $e->getMessage();
        $this->importDiff = ['new' => [], 'unchanged' => [], 'conflicts' => []];
    }
}

public function confirmImport(AliasYamlImporter $importer, CurrentUser $currentUser): void {
    $allEntries = array_merge($this->importDiff['new'], $this->importDiff['conflicts']);
    $changed = $importer->apply($currentUser->user(), $allEntries, $this->conflictResolutions);
    $this->flashMessage = sprintf('Imported %d changes.', $changed);
    $this->importFile = null;
    $this->importDiff = ['new' => [], 'unchanged' => [], 'conflicts' => []];
    $this->conflictResolutions = [];
}
```

Methods (all method-DI):
```
#[\Livewire\Attributes\On('rename-counterparty:open')]
public function open(string $raw, int $rowIndex, ?int $currentCategoryId = null, PatternGeneralizer $generalizer): void {
    $this->raw = $raw;
    $this->rowIndex = $rowIndex;
    $this->friendly = '';
    $this->remember = true;
    $this->categoryHint = $currentCategoryId;
    $this->generalized = $generalizer->generalize($raw);
    $this->dispatch('modal-show', name: 'rename-counterparty');
}

public function updatedGeneralized(PatternGeneralizer $generalizer): void {
    // user-edited the generalized pattern — keep it as-is; do NOT auto-regenerate
}

public function save(
    CreateMerchantAlias $createAlias,
    \Modules\Categorization\Public\Actions\CreateCategorizationRule $createRule,
    \Modules\Core\Public\Contracts\CurrentUser $currentUser,
): void {
    $this->validate(['friendly' => 'required|string|min:1|max:255', 'raw' => 'required|string|min:1', 'generalized' => 'nullable|string|max:255']);
    $user = $currentUser->user();
    if ($this->remember) {
        ($createAlias)($user, $this->raw, $this->generalized !== '' ? $this->generalized : null, $this->friendly);
    }
    if ($this->categoryHint !== null) {
        ($createRule)($user, ['field' => 'description', 'match' => 'contains', 'value' => $this->generalized, 'category_id' => $this->categoryHint, 'active' => true]);
    }
    $this->dispatch('rename-counterparty:saved', rowIndex: $this->rowIndex, friendlyName: $this->friendly);
    $this->dispatch('modal-hide', name: 'rename-counterparty');
}

public function render(\Illuminate\Contracts\View\Factory $views): \Illuminate\Contracts\View\View {
    return $views->make('import::livewire.rename-counterparty-popover');
}
```

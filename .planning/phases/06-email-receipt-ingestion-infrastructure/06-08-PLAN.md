      public function emailScanHealth(User $user): ?EmailScanHealthTile {
          $rows = $this->db->connection()
              ->table('inboxes')
              ->leftJoin('inbox_scan_state', function ($join) {
                  $join->on('inbox_scan_state.inbox_id', '=', 'inboxes.id')->where('inbox_scan_state.folder', 'INBOX');
              })
              ->where('inboxes.user_id', $user->id)
              ->orderBy('inboxes.created_at', 'asc')
              ->select([
                  'inboxes.id',
                  'inboxes.provider',
                  'inboxes.email',
                  'inbox_scan_state.status',
                  'inbox_scan_state.last_scan_at',
              ])
              ->get();

          if ($rows->isEmpty()) return null;   // hide tile entirely

          // Compute overall status: 'reauth' if any inbox needs_reauth; else 'stale' if any > 24h or never scanned; else 'healthy'.
          $now = $this->clock->now();   // inject Clock — if not in this class' DI, add it; the existing nextIcsSettlement() method should already use it
          $overall = 'healthy';
          $lines = [];
          foreach ($rows->take(3) as $row) {
              $status = (string) ($row->status ?? 'idle');
              $lastScan = $row->last_scan_at ? new \DateTimeImmutable((string) $row->last_scan_at) : null;
              $lineStatus = 'healthy';
              if ($status === 'needs_reauth') {
                  $lineStatus = 'reauth';
                  $overall = 'reauth';
              } elseif ($lastScan === null) {
                  $lineStatus = 'stale';
                  if ($overall !== 'reauth') $overall = 'stale';
              } elseif ($lastScan->getTimestamp() < $now->getTimestamp() - 86400) {
                  $lineStatus = 'stale';
                  if ($overall !== 'reauth') $overall = 'stale';
              }
              $emailLocalPart = explode('@', (string) $row->email)[0];
              $lines[] = new InboxHealthLine(
                  provider: (string) $row->provider,
                  emailLocalPart: substr($emailLocalPart, 0, 12),
                  lastScanAt: $lastScan,
                  status: $lineStatus,
              );
          }
          $overflowCount = max(0, $rows->count() - 3);
          return new EmailScanHealthTile(lines: $lines, overallStatus: $overall, overflowCount: $overflowCount);
      }

      public function scanNow(int $inboxId, CurrentUser $cu, InboxQuery $inboxQuery, \Illuminate\Contracts\Bus\Dispatcher $bus): void {
          $user = $cu->user();
          $health = $inboxQuery->findForUser($inboxId, $user);
          if ($health === null) throw new NotFoundHttpException('Inbox not found.');
          if (in_array($health->status, ['backfilling', 'scanning'], true)) {
              $this->dispatch('toast', message: 'Scan already in progress.');
              return;
          }
          $bus->dispatch(new IncrementalScanJob($inboxId));
          $this->dispatch('toast', message: 'Scan started.');
      }

      // reconnect just redirects — no Livewire action needed beyond the link rendering; the Blade renders an <a href="{{ route('oauth.connect', ...) }}?inbox_id={{ $id }}"> directly. But we add a method anyway for testability:
      public function reconnect(int $inboxId, CurrentUser $cu, InboxQuery $inboxQuery, RedirectFactory $redirector) {
          $user = $cu->user();
          $health = $inboxQuery->findForUser($inboxId, $user);
          if ($health === null) throw new NotFoundHttpException('Inbox not found.');
          return $redirector->route('oauth.connect', ['provider' => $health->provider]) . '?inbox_id=' . $inboxId;

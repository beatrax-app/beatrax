## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| DiscoveryScanJob query → provider | Broad query touches many messages; bandwidth-bounded by daily cadence |
| launchd plist → macOS service | The plist runs `php artisan horizon` indefinitely; misconfiguration leads to crash-loop |
| User input → discovered_sender promotion | Promoting a sender adds it to the primary scan filter; could be used to harvest unrelated mail if the panel accepts crafted sender_email values |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-09-01 | Denial of service | Broad discovery query exhausts Gmail quota | mitigate | Daily cadence + $top=100 + Sleep::seconds(2) between pages; on rate_limit → DiscoveryScanJob silently aborts (no .eml blobs to persist; minimal cost to re-run tomorrow) |
| T-06-09-02 | Tampering | Promotion adds a hostile sender to known_senders | mitigate | PromoteDiscoveredSender inserts from an existing discovered_senders row — the sender_email value was provider-supplied, not user-typed; cross-user 404 prevents promoting another user's discovered row |
| T-06-09-03 | Tampering | launchd plist crash-loop on PHP_BINARY mismatch | mitigate | ThrottleInterval=10 in the plist; InstallCommand substitutes PHP_BINARY at install time; README documents re-run after Herd upgrade |
| T-06-09-04 | Information disclosure | discovered_senders row leaks across users | mitigate | UNIQUE (user_id, inbox_id, sender_email) constraint from Plan 02 + cross-user 404 in Public Actions |

    handle(DatabaseManager $db, Clock $clock, GmailApiClient $gmail, GraphApiClient $graph, OAuthSecretsRepository $secrets, KnownSenderQuery $senderQuery): void:

      $user = \App\Models\User::query()->where('id', $this->userId)->firstOrFail();

      // 1. Read excluded senders: every known_senders pattern + every dismissed discovered_senders sender for this user
      $excludeSenders = array_map(fn($s) => $s->emailPattern, $senderQuery->all($user));
      $dismissedSenders = $db->connection()->table('discovered_senders')
          ->where('user_id', $this->userId)->whereIn('state', ['dismissed', 'added'])
          ->pluck('sender_email')->toArray();
      $allExcludes = array_unique(array_merge($excludeSenders, $dismissedSenders));

      // 2. Enumerate user's connected inboxes
      $inboxes = $db->connection()->table('inboxes')->where('user_id', $this->userId)->get();
      if ($inboxes->isEmpty()) return;

      $keywords = ['receipt', 'factuur', 'betaling', 'invoice', 'order', 'bevestiging'];

      foreach ($inboxes as $inbox) {
          $inboxId = (int) $inbox->id;
          $provider = (string) $inbox->provider;
          try {
              if ($provider === 'gmail') {
                  $page = $gmail->listDiscoveryCandidates($inboxId, $keywords, $allExcludes);
                  $messages = $page['messages'] ?? [];
              } elseif ($provider === 'microsoft') {
                  $page = $graph->listDiscoveryCandidatesPaged($inboxId, $keywords, $allExcludes, null);
                  $messages = $page['messages'] ?? [];
              } else {
                  continue;
              }

              foreach ($messages as $msg) {
                  $sender = strtolower($msg['fromAddress'] ?? '');
                  if ($sender === '') continue;
                  $senderName = $msg['fromName'] ?? null;
                  $internalDate = new \DateTimeImmutable($msg['receivedDateTime'] ?? $msg['internalDate'] ?? 'now');
                  // Skip already-known senders defensively (in case the q= filter didn't exclude)
                  if (in_array($sender, $allExcludes, true)) continue;
                  // Skip senders matching @-suffix excludes
                  foreach ($allExcludes as $excludePattern) {
                      if (str_starts_with(strtolower($excludePattern), '@') && str_ends_with($sender, strtolower($excludePattern))) {
                          continue 2;
                      }
                  }

                  // Upsert discovered_senders row
                  $existing = $db->connection()->table('discovered_senders')
                      ->where('user_id', $this->userId)->where('inbox_id', $inboxId)->where('sender_email', $sender)->first();
                  if ($existing === null) {
                      $db->connection()->table('discovered_senders')->insert([
                          'user_id' => $this->userId,
                          'inbox_id' => $inboxId,
                          'sender_email' => $sender,
                          'sender_name' => $senderName,
                          'occurrence_count' => 1,
                          'last_seen_at' => $internalDate->format('Y-m-d H:i:s'),
                          'state' => 'candidate',
                          'created_at' => $clock->now()->toDateTimeString(),
                          'updated_at' => $clock->now()->toDateTimeString(),
                      ]);
                  } else {
                      // Only increment count + last_seen if the existing state is 'candidate' — don't resurface dismissed/added
                      if ($existing->state === 'candidate') {
                          $db->connection()->table('discovered_senders')
                              ->where('id', $existing->id)
                              ->update([
                                  'occurrence_count' => $existing->occurrence_count + 1,
                                  'last_seen_at' => $internalDate->format('Y-m-d H:i:s'),
                                  'sender_name' => $senderName ?? $existing->sender_name,
                                  'updated_at' => $clock->now()->toDateTimeString(),
                              ]);
                      }
                  }

                  // CRITICAL INVARIANT: DiscoveryScanJob NEVER calls $gmail->getRawMessage or $graph->getRawMessage or $blobStore->put.
                  // No .eml blobs are persisted from the discovery loop (D-121 + integration test asserts this).
              }
          } catch (RateLimitedException $e) {
              // Discovery is daily; rate limit means abort silently and retry tomorrow.
              return;
          } catch (\Throwable $e) {
              // Discovery failures are best-effort — log nothing (no facade access to Log in module code); silently continue.
              continue;
          }
      }

      public function __invoke(int $discoveredSenderId, User $user): void {
          $this->db->connection()->transaction(function () use ($discoveredSenderId, $user) {
              $this->db->connection()->statement('PRAGMA busy_timeout = 5000');
              $row = $this->db->connection()->table('discovered_senders')
                  ->where('id', $discoveredSenderId)->where('user_id', $user->id)
                  ->lockForUpdate()->first();
              if ($row === null) throw new NotFoundHttpException('Discovered sender not found.');
              if ($row->state !== 'candidate') return;  // idempotent — already promoted or dismissed
              $now = $this->clock->now()->toDateTimeString();
              // Insert into known_senders with source='user'
              $this->db->connection()->table('known_senders')->insert([
                  'user_id' => $user->id,
                  'email_pattern' => $row->sender_email,
                  'label' => $row->sender_name ?: $row->sender_email,
                  'source' => 'user',
                  'added_at' => $now,
                  'created_at' => $now,
                  'updated_at' => $now,
              ]);

      $plists = ['horizon', 'scheduler'];
      if (! $this->option('without-redis')) $plists[] = 'redis';
      $substitutions = [
          '{{ABS_PHP_BINARY}}' => PHP_BINARY,
          '{{ABS_PROJECT_ROOT}}' => base_path(),
      ];
      foreach ($plists as $name) {
          $sourcePath = base_path('deploy/launchd/com.diederik.' . $name . '.plist');
          $targetPath = $launchAgentsDir . '/com.diederik.' . $name . '.plist';
          $template = $this->files->get($sourcePath);
          $rendered = strtr($template, $substitutions);
          $this->files->put($targetPath, $rendered);
          $this->info("Wrote {$targetPath}");
          // bootstrap via launchctl
          $uid = posix_getuid();
          $cmd = "launchctl bootstrap gui/{$uid} " . escapeshellarg($targetPath);
          passthru($cmd, $exitCode);
          if ($exitCode === 0) {
              $this->info("Loaded com.diederik.{$name}");
          } else {
              $this->warn("launchctl bootstrap exited {$exitCode} for com.diederik.{$name} (may already be loaded; check `launchctl list | grep diederik`)");
          }
      }
      return self::SUCCESS;

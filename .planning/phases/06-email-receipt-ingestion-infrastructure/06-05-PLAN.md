      public function __construct(
          private readonly OAuthSecretsRepository $secrets,
          private readonly GoogleOAuthProvider $oauth,
          private readonly Clock $clock,
      ) {}

      Private ensureFreshAccessToken(int $inboxId): string —
        $creds = $this->secrets->loadInbox($inboxId);
        if ($creds === null) throw RuntimeException("No credentials for inbox {$inboxId}");
        if ($creds->expiresAt === null || $creds->expiresAt->getTimestamp() < $this->clock->now()->getTimestamp() + 60) {
            // refresh
            $fresh = $this->oauth->refreshAccessToken($creds->refreshToken);
            // Gmail does NOT rotate refresh tokens single-use — only the access token changes; but rotateRefreshToken handles the no-op gracefully via $newRefreshToken === $old.
            $this->secrets->rotateRefreshToken($inboxId, $fresh->refreshToken ?? $creds->refreshToken, $fresh->accessToken, $fresh->expiresAt);
            return $fresh->accessToken;
        }
        return $creds->accessToken ?? throw new RuntimeException("Fresh token cache empty");

      private function getGmailService(int $inboxId): \Google\Service\Gmail —
        $accessToken = $this->ensureFreshAccessToken($inboxId);
        $client = new \Google\Client();
        $client->setAccessToken(['access_token' => $accessToken, 'expires_in' => 3600]);
        return new \Google\Service\Gmail($client);

      Public listSenderMessages(int $inboxId, array $senderPatterns, ?string $pageToken): array —
        $gmail = $this->getGmailService($inboxId);
        $q = 'from:(' . implode(' OR ', $senderPatterns) . ')';
        $params = ['q' => $q, 'maxResults' => 100];
        if ($pageToken !== null) $params['pageToken'] = $pageToken;
        try {
            $response = $gmail->users_messages->listUsersMessages('me', $params);
        } catch (\Google\Service\Exception $e) {
            $reason = $e->getErrors()[0]['reason'] ?? '';
            if (in_array($reason, ['rateLimitExceeded','userRateLimitExceeded','dailyLimitExceeded'], true)) {
                throw new RateLimitedException(retryAfterSeconds: 60);
            }
            throw $e;
        }
        return [
            'messages' => array_map(fn($m) => ['id' => $m->getId(), 'threadId' => $m->getThreadId()], $response->getMessages() ?? []),
            'nextPageToken' => $response->getNextPageToken(),
            'resultSizeEstimate' => $response->getResultSizeEstimate(),
        ];

      Public getRawMessage(int $inboxId, string $providerMessageId): string —
        $gmail = $this->getGmailService($inboxId);
        try {
            $msg = $gmail->users_messages->get('me', $providerMessageId, ['format' => 'raw']);
        } catch (\Google\Service\Exception $e) { /* same rate-limit map */ }
        // Gmail's raw is base64url; decode:
        $raw = $msg->getRaw();
        return base64_decode(strtr($raw, '-_', '+/') . str_repeat('=', (4 - strlen($raw) % 4) % 4));

      Public listHistory(int $inboxId, string $startHistoryId): array —
        try { $response = $gmail->users_history->listUsersHistory('me', ['startHistoryId' => $startHistoryId]); }
        catch (\Google\Service\Exception $e) {
            if ($e->getCode() === 404) throw new CursorExpiredException(provider: 'gmail');
            $reason = ...; rate-limit map as above;
        }
        return ['history' => $response->getHistory() ?? [], 'historyId' => $response->getHistoryId()];

    2. **`ThisPeriodAtAGlanceQuery`** implementation exactly per `01-RESEARCH.md §This-Period-At-A-Glance Query — The dashboard query`:
       ```php
       namespace Modules\Ledger\Public\Services;

       use Illuminate\Database\DatabaseManager;
       use Modules\Core\Models\User;
       use Modules\Ledger\Models\Transaction;
       use Modules\Ledger\Public\Dto\DashboardSummary;
       use Modules\Ledger\Public\Dto\Period;
       use Modules\Ledger\Public\ValueObjects\Money;

       final class ThisPeriodAtAGlanceQuery
       {
           public function __construct(
               private readonly DatabaseManager $db,
               private readonly TopCategoriesByPeriodQuery $topCategoriesQuery,
               private readonly TransactionListQuery $listQuery,
           ) {}

           public function for(User $user, Period $period): DashboardSummary
           {
               $totalCount = Transaction::query()->where('user_id', $user->id)->count();
               if ($totalCount === 0) {
                   return new DashboardSummary(
                       period: $period,
                       inflow:  Money::ofMinor(0, 'EUR'),
                       outflow: Money::ofMinor(0, 'EUR'),
                       net:     Money::ofMinor(0, 'EUR'),
                       topCategories: [],
                       recentTransactions: [],
                       uncategorizedCount: 0,
                       isFirstRun: true,
                   );
               }

               $startStr = $period->start->toDateString();
               $endStr = $period->endExclusive->toDateString();

               $row = $this->db->connection()
                   ->table('transactions')
                   ->where('user_id', $user->id)
                   ->where('posted_at', '>=', $startStr)
                   ->where('posted_at', '<', $endStr)
                   ->selectRaw('
                       COALESCE(SUM(CASE WHEN amount_minor > 0 THEN amount_minor ELSE 0 END), 0) AS inflow_minor,
                       COALESCE(SUM(CASE WHEN amount_minor < 0 THEN -amount_minor ELSE 0 END), 0) AS outflow_minor,
                       COALESCE(SUM(amount_minor), 0) AS net_minor
                   ')
                   ->first();

               $uncategorized = Transaction::query()
                   ->where('user_id', $user->id)
                   ->whereNull('category_id')
                   ->count();

               $recent = $this->listQuery->recent($user, daysBack: 90, limit: 10);

               return new DashboardSummary(
                   period: $period,
                   inflow:  Money::ofMinor((int) $row->inflow_minor, 'EUR'),
                   outflow: Money::ofMinor((int) $row->outflow_minor, 'EUR'),
                   net:     Money::ofMinor((int) $row->net_minor, 'EUR'),
                   topCategories: $this->topCategoriesQuery->for($user, $period, limit: 5),
                   recentTransactions: $recent->rows,
                   uncategorizedCount: $uncategorized,
                   isFirstRun: false,
               );
           }
       }
       ```

    3. **`TopCategoriesByPeriodQuery`** per `01-RESEARCH.md §Top categories per period`:
       ```php
       namespace Modules\Ledger\Public\Services;

       use Illuminate\Database\DatabaseManager;
       use Modules\Core\Models\User;
       use Modules\Ledger\Models\Category;
       use Modules\Ledger\Public\Dto\Period;
       use Modules\Ledger\Public\Dto\TopCategoryRow;
       use Modules\Ledger\Public\ValueObjects\Money;

       final class TopCategoriesByPeriodQuery
       {
           public function __construct(private readonly DatabaseManager $db) {}

           /** @return array<TopCategoryRow> */
           public function for(User $user, Period $period, int $limit = 5): array
           {
               $rows = $this->db->connection()
                   ->table('transactions')
                   ->where('user_id', $user->id)
                   ->where('amount_minor', '<', 0)
                   ->where('posted_at', '>=', $period->start->toDateString())
                   ->where('posted_at', '<', $period->endExclusive->toDateString())
                   ->whereNotNull('category_id')
                   ->groupBy('category_id')
                   ->orderByRaw('SUM(-amount_minor) DESC')
                   ->limit($limit)
                   ->get(['category_id', $this->db->connection()->raw('SUM(-amount_minor) AS spend_minor')]);

               if ($rows->isEmpty()) return [];

               $total = (int) $rows->sum('spend_minor');
               $byId = Category::query()->whereIn('id', $rows->pluck('category_id'))->get()->keyBy('id');

               $result = [];
               foreach ($rows as $r) {
                   $cat = $byId[$r->category_id] ?? null;
                   if ($cat === null) continue;
                   $result[] = new TopCategoryRow(
                       categoryId: (int) $r->category_id,
                       name: $this->fullPath($cat),
                       spend: Money::ofMinor((int) $r->spend_minor, 'EUR'),
                       percentageOfTotal: $total > 0 ? (float) $r->spend_minor / (float) $total : 0.0,
                   );
               }
               return $result;
           }

           private function fullPath(Category $cat): string
           {
               $parts = [$cat->name];
               $p = $cat->parent;
               while ($p !== null) { array_unshift($parts, $p->name); $p = $p->parent; }
               return implode(' / ', $parts);
           }
       }
       ```

    4. **`TransactionListQuery`** — cursor-based pagination:
       ```php
       namespace Modules\Ledger\Public\Services;

       use Carbon\CarbonImmutable;
       use Modules\Core\Models\User;
       use Modules\Core\Public\Contracts\Clock;
       use Modules\Ledger\Models\Transaction;
       use Modules\Ledger\Public\Dto\TransactionListPage;
       use Modules\Ledger\Public\Dto\TransactionRowDto;
       use Modules\Ledger\Public\ValueObjects\Money;

       final class TransactionListQuery
       {
           public function __construct(private readonly Clock $clock) {}

           public function recent(User $user, int $daysBack = 90, ?int $cursorId = null, int $limit = 50): TransactionListPage
           {
               $cutoff = $this->clock->now()->subDays($daysBack);
               $q = Transaction::query()
                   ->where('user_id', $user->id)
                   ->where('posted_at', '>=', $cutoff->toDateString())
                   ->orderByDesc('posted_at')
                   ->orderByDesc('id')
                   ->limit($limit + 1);

               if ($cursorId !== null) $q->where('id', '<', $cursorId);

               return $this->buildPage($q->get(), $limit);
           }

           public function fullHistory(User $user, ?int $cursorId = null, int $limit = 50): TransactionListPage
           {
               $q = Transaction::query()
                   ->where('user_id', $user->id)
                   ->orderByDesc('posted_at')
                   ->orderByDesc('id')
                   ->limit($limit + 1);

               if ($cursorId !== null) $q->where('id', '<', $cursorId);

               return $this->buildPage($q->get(), $limit);
           }

           private function buildPage(\Illuminate\Support\Collection $rows, int $limit): TransactionListPage
           {
               $hasMore = $rows->count() > $limit;
               $sliced = $rows->take($limit);

               $byCategoryId = $sliced->pluck('category_id')->filter()->unique();
               $categories = \Modules\Ledger\Models\Category::query()->whereIn('id', $byCategoryId)->get()->keyBy('id');

               $dtos = $sliced->map(fn (Transaction $t) => new TransactionRowDto(
                   id: (int) $t->id,
                   bookedAt: $t->booked_at->format('d-m-Y'),
                   counterpartyName: $t->counterparty_name,
                   categoryId: $t->category_id,
                   categoryName: $t->category_id ? ($categories[$t->category_id]->name ?? null) : null,
                   amount: Money::ofMinor((int) $t->amount_minor, (string) $t->currency),
               ));

               return new TransactionListPage(
                   rows: $dtos->all(),
                   hasMore: $hasMore,
                   nextCursorId: $hasMore ? (int) $sliced->last()->id : null,
               );
           }
       }
       ```

       ```php
       namespace Modules\Core\Internal\Http\Livewire;

       use Carbon\CarbonImmutable;
       use Illuminate\Contracts\Routing\UrlGenerator;
       use Illuminate\Contracts\View\Factory as ViewFactory;
       use Illuminate\Contracts\View\View;
       use Illuminate\Http\RedirectResponse;
       use Livewire\Component;
       use Modules\Core\Public\Contracts\CurrentUser;
       use Modules\Ledger\Public\Dto\DashboardSummary;
       use Modules\Ledger\Public\Dto\Period;
       use Modules\Ledger\Public\Services\PeriodQuery;
       use Modules\Ledger\Public\Services\ThisPeriodAtAGlanceQuery;

       final class Dashboard extends Component
       {
           public ?string $periodStartStr = null;   // 'YYYY-MM-DD' anchor, null = current

           public function boot(
               private CurrentUser $currentUser,
               private PeriodQuery $periods,
               private ThisPeriodAtAGlanceQuery $glance,
               private UrlGenerator $urls,
           ): void {}

           public function render(ViewFactory $views): View|RedirectResponse
           {
               $user = $this->currentUser->user();
               $period = $this->periodStartStr
                   ? $this->periods->containing(CarbonImmutable::parse($this->periodStartStr))
                   : $this->periods->current();
               $summary = $this->glance->for($user, $period);

               if ($summary->isFirstRun) {
                   return new RedirectResponse($this->urls->route('imports.new'));
               }

               return $views->make('core::livewire.dashboard', ['summary' => $summary]);
           }

           public function previousPeriod(): void
           {
               $current = $this->periodStartStr
                   ? $this->periods->containing(CarbonImmutable::parse($this->periodStartStr))
                   : $this->periods->current();
               $this->periodStartStr = $this->periods->previous($current)->start->toDateString();
           }

           public function nextPeriod(): void
           {
               $current = $this->periodStartStr
                   ? $this->periods->containing(CarbonImmutable::parse($this->periodStartStr))
                   : $this->periods->current();
               $this->periodStartStr = $this->periods->next($current)->start->toDateString();
           }

           public function today(): void { $this->periodStartStr = null; }
       }
       ```

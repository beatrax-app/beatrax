       use Illuminate\Contracts\Events\Dispatcher;
       use Modules\Categorization\Public\Contracts\AssignsCategory;
       use Modules\Categorization\Public\Events\TransactionCategorized;

       final class AssignCategory implements AssignsCategory
       {
           public function __construct(
               private readonly UpdatesTransactionCategory $updater,

           ) {}

           public function __invoke(int $transactionId, ?int $categoryId, User $user): int
           {
               $affected = ($this->updater)($transactionId, $categoryId, $user);
               if ($affected > 0) {
                   $this->events->dispatch(new TransactionCategorized(

                       categoryId: $categoryId,
                       userId: (int) $user->id,
                   ));
               }
               return $affected;
           }
       }

    6. **`UncategorizedTriageQuery`** per `01-RESEARCH.md §Triage query`:
       ```php
       namespace Modules\Categorization\Public\Services;

       use Modules\Categorization\Public\Dto\TriageBatch;
       use Modules\Categorization\Public\Dto\TriageRow;
       use Modules\Core\Models\User;
       use Modules\Ledger\Models\Transaction;

       final class UncategorizedTriageQuery
       {
           public function for(User $user, int $limit = 50, ?int $cursorId = null): TriageBatch
           {
               $q = Transaction::query()
                   ->where('user_id', $user->id)
                   ->whereNull('category_id')
                   ->orderByDesc('posted_at')
                   ->orderByDesc('id')
                   ->limit($limit + 1);

               if ($cursorId !== null) $q->where('id', '<', $cursorId);

               $rows = $q->get();
               $hasMore = $rows->count() > $limit;
               $sliced = $rows->take($limit);

               $dtos = $sliced->map(fn (Transaction $t) => new TriageRow(
                   transactionId: (int) $t->id,
                   bookedAt: $t->booked_at->format('d-m-Y'),
                   counterpartyName: $t->counterparty_name,
                   amountMinor: (int) $t->amount_minor,
                   currency: (string) $t->currency,
                   description: $t->description,
               ))->all();

               return new TriageBatch(
                   rows: $dtos,
                   hasMore: $hasMore,
                   nextCursorId: $hasMore ? (int) $sliced->last()->id : null,
               );
           }
       }
       ```

       namespace Modules\Categorization\Internal\Http\Livewire;

       use Illuminate\Contracts\View\Factory as ViewFactory;
       use Illuminate\Contracts\View\View;
       use Livewire\Component;
       use Modules\Categorization\Public\Contracts\AssignsCategory;
       use Modules\Categorization\Public\Dto\CategoryOption;
       use Modules\Categorization\Public\Services\UncategorizedTriageQuery;
       use Modules\Core\Public\Contracts\CurrentUser;
       use Modules\Ledger\Models\Category;

       final class TriageInbox extends Component
       {
           /** @var array<int, ?int> map of transactionId => pending categoryId */
           public array $pending = [];

           public function boot(
               private CurrentUser $currentUser,
               private UncategorizedTriageQuery $triage,
               private AssignsCategory $assign,
           ): void {}

           public function selectForRow(int $transactionId, ?int $categoryId): void
           {
               $this->pending[$transactionId] = $categoryId;
           }

           public function save(): void
           {
               $user = $this->currentUser->user();
               foreach ($this->pending as $txId => $catId) {
                   ($this->assign)($txId, $catId, $user);
               }
               $this->pending = [];
           }

           public function clearPending(): void { $this->pending = []; }

           public function render(ViewFactory $views): View
           {
               $user = $this->currentUser->user();
               $batch = $this->triage->for($user);

               $allCategories = Category::query()
                   ->orderBy('display_order')
                   ->get()
                   ->map(fn (Category $c) => new CategoryOption(
                       id: (int) $c->id,
                       path: $c->parent_id ? ($c->parent->name . ' / ' . $c->name) : $c->name,
                       displayOrder: (int) $c->display_order,
                   ))
                   ->all();

               $topNine = array_slice($allCategories, 0, 9);            // TopNCategories assumption — display_order proxy

               return $views->make('categorization::livewire.triage-inbox', [
                   'batch' => $batch,
                   'categories' => $allCategories,
                   'topNine' => $topNine,
               ]);
           }
       }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Livewire wire:change → AssignCategory | User-supplied `categoryId` flows to the action. The action filters by `user_id` so cross-user assignments are blocked at the DB layer. |
| Listener → seeder | UserInstalled event fires from InstallCommand (run by host user via CLI); listener invokes seeder. No request-time exposure. |
| Triage page → DB | `UncategorizedTriageQuery::for($user)` filters by `$user->id` explicitly. `UserScope` on Transaction adds the same WHERE. |

## STRIDE Threat Register (ASVS L1)

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-01 | E (Elevation of Privilege) | User A assigns a category to user B's transaction | mitigate | `UpdateTransactionCategory::__invoke` filters `Transaction::where('id', $id)->where('user_id', $user->id)->update(...)`. Returns 0 affected rows; AssignCategory does not fire the event when affected===0. |
| T-07-02 | T (Tampering) | Forged `categoryId` referencing a non-existent or other-user's category | accept (Phase 1) | Categories are USER-scoped via `BelongsToUser`, but seeded categories have `user_id = NULL` (global). A user could assign a category they don't own; Phase 1 treats categories as shared/global. Phase 7 (CAT-02 + CAT-04) may need to formalize user ownership. Flagged in SUMMARY. |
| T-07-03 | I (Information Disclosure) | Triage page enumerates other users' transactions | mitigate | Same `UserScope` + explicit `where('user_id', $user->id)` in `UncategorizedTriageQuery`. |
| T-07-04 | T (Tampering) | XSS via category name | mitigate | Categories are seeder-controlled (no user input in Phase 1). Future user-defined names (CAT-04, Phase 7) will require Blade escaping; already automatic with `{{ }}`. |
| T-07-05 | T (Tampering) | Listener mis-fire — UserInstalled event seeds categories twice | mitigate | `DefaultCategoryTreeSeeder` uses `updateOrCreate(['slug' => ...])` — idempotent. Re-running install does not duplicate categories. Tested. |
</threat_model>

interface SourceAdapter
{
    /** Stable format identifier; e.g. 'asn-csv', 'asn-camt053' (Phase 2). */
    public function format(): string;

    /**
     * Stream parsed rows from the source file. The implementation MUST be a Generator
     * (or otherwise lazy) so memory stays bounded for large files.
     *
     * @return \Generator<SourceTransactionDto>
     */
    public function parse(string $localPath, AccountResolver $accounts): \Generator;
}

interface AccountResolver
{
    /**
     * Map IBAN → existing Account or 'Unknown'. The wizard (Plan 05) supplies a resolver
     * that returns Known(Account) for any IBAN already in the DB, and Unknown(iban)
     * otherwise so the wizard can pause for the user to name the account.
     */
    public function resolve(string $iban): AccountResolution;
}

namespace Modules\Ingestion\Public\Dto;

use Spatie\LaravelData\Data;

abstract class AccountResolution extends Data {
    public static function known(int $accountId): KnownAccount { return new KnownAccount($accountId); }
    public static function unknown(string $iban): UnknownAccount { return new UnknownAccount($iban); }
}
final class KnownAccount extends AccountResolution { public function __construct(public readonly int $accountId) {} }
final class UnknownAccount extends AccountResolution { public function __construct(public readonly string $iban) {} }

    public function __construct(
        public readonly \Carbon\CarbonImmutable $bookedAt,

        public readonly string $ownIban,

}

       namespace Modules\Ingestion\Internal\Adapters\Asn;

       use Modules\Ingestion\Public\Exceptions\InvalidAmountException;

       final class AsnAmountParser
       {
           public function parseMinor(string $raw): int
           {
               // ASN format: period-decimal, signed, two digits — e.g. "-12.34", "+0.29", "1000.00"
               $normalized = str_replace(['+', ' ', "\u{A0}"], '', trim($raw));

               if (preg_match('/^(-?)(\d+)\.(\d{2})$/', $normalized, $m) !== 1) {
                   throw new InvalidAmountException("Cannot parse ASN amount: '{$raw}'");
               }
               $sign = $m[1] === '-' ? -1 : 1;
               $whole = (int) $m[2];
               $fractional = (int) $m[3];
               return $sign * ($whole * 100 + $fractional);
           }
       }

       ```php
       namespace Modules\Ingestion\Public\Services;

       use Modules\Ingestion\Internal\Adapters\Asn\AsnCsvHeaderProfile;
       use Modules\Ingestion\Public\Exceptions\SniffMismatchException;

       final class HeaderSniffer
       {
           public function sniff(string $localPath, string $declaredFormat): SniffResult
           {
               if (!is_file($localPath) || !is_readable($localPath)) {
                   throw new SniffMismatchException("File not readable: {$localPath}");
               }

               // Read first 8 KB to inspect
               $handle = fopen($localPath, 'rb');
               $head = fread($handle, 8192);
               fclose($handle);

               switch ($declaredFormat) {
                   case AsnCsvHeaderProfile::FORMAT:
                       return $this->sniffAsnCsv($localPath, $head);
                   default:
                       throw new SniffMismatchException("Unsupported sniff target: {$declaredFormat}");
               }
           }

           private function sniffAsnCsv(string $path, string $head): SniffResult
           {
               // Validate extension
               if (!preg_match('/\.csv$/i', $path)) {
                   throw new SniffMismatchException("That file doesn't look like a CSV. Drop in the ASN CSV export you downloaded from the ASN portal.");
               }

               // First-row signature — date prefix per AsnCsvHeaderProfile::FIRST_ROW_PATTERN
               $firstLine = strtok($head, "\r\n") ?: '';
               if (preg_match(AsnCsvHeaderProfile::FIRST_ROW_PATTERN, $firstLine) !== 1
                   && !AsnCsvHeaderProfile::HAS_HEADER) {
                   throw new SniffMismatchException("This CSV doesn't match the expected ASN column layout. If ASN changed their export format, file an issue.");
               }

               // Column count
               $delim = AsnCsvHeaderProfile::DELIMITER;
               $columns = str_getcsv($firstLine, $delim);
               if (count($columns) !== AsnCsvHeaderProfile::EXPECTED_COLUMN_COUNT) {
                   throw new SniffMismatchException(
                       "Expected " . AsnCsvHeaderProfile::EXPECTED_COLUMN_COUNT . " columns, got " . count($columns) . "."
                   );
               }

               return new SniffResult(
                   format: AsnCsvHeaderProfile::FORMAT,
                   delimiter: $delim,
                   hasHeader: AsnCsvHeaderProfile::HAS_HEADER,
                   encoding: AsnCsvHeaderProfile::SOURCE_ENCODING,
                   columnCount: count($columns),
               );
           }
       }
       ```

       ```php
       namespace Modules\Ingestion\Internal\Adapters\Asn;

       use Carbon\CarbonImmutable;
       use League\Csv\CharsetConverter;
       use League\Csv\Reader;
       use League\Csv\Statement;
       use Modules\Ingestion\Public\Contracts\AccountResolver;
       use Modules\Ingestion\Public\Contracts\SourceAdapter;
       use Modules\Ingestion\Public\Dto\SourceTransactionDto;
       use Modules\Ingestion\Public\Services\HeaderSniffer;

       final class AsnCsvAdapter implements SourceAdapter
       {
           public function __construct(
               private readonly AsnAmountParser $amounts,
               private readonly HeaderSniffer $sniffer,
           ) {}

           public function format(): string { return AsnCsvHeaderProfile::FORMAT; }

           public function parse(string $localPath, AccountResolver $accounts): \Generator
           {
               $this->sniffer->sniff($localPath, AsnCsvHeaderProfile::FORMAT);   // throws on bad header

               $reader = Reader::createFromPath($localPath, 'r');
               $reader->setDelimiter(AsnCsvHeaderProfile::DELIMITER);
               $reader->setEscape('');                                             // PHP 8.4+ deprecation guard
               if (AsnCsvHeaderProfile::HAS_HEADER) {
                   $reader->setHeaderOffset(0);
               } else {
                   $reader->setHeaderOffset(null);
               }
               CharsetConverter::addTo($reader, AsnCsvHeaderProfile::SOURCE_ENCODING, 'utf-8');

               $i = 0;
               foreach ((new Statement())->process($reader) as $row) {
                   // $row is array<int,string> when HAS_HEADER=false, array<string,string> when true.
                   // Normalize to int-indexed for predictable column lookups:
                   $r = array_values((array) $row);

                   try {
                       $bookedAt = CarbonImmutable::createFromFormat('!d-m-Y', $r[AsnCsvColumnMap::POSTED_DATE]);
                       $valueDate = CarbonImmutable::createFromFormat('!d-m-Y', $r[AsnCsvColumnMap::VALUE_DATE]);
                       $amountMinor = $this->amounts->parseMinor($r[AsnCsvColumnMap::AMOUNT]);
                   } catch (\Throwable $e) {
                       // Defensive: skip malformed rows. The Wizard (Plan 05) renders ERROR for these.
                       // Adapter still yields a DTO with as-much-as-possible info so the wizard can show the error.
                       throw new \Modules\Ingestion\Public\Exceptions\InvalidAmountException(
                           "Row {$i}: " . $e->getMessage(),
                           previous: $e
                       );
                   }

                   $ownIban = (string) $r[AsnCsvColumnMap::OWN_IBAN];
                   $resolution = $accounts->resolve($ownIban);                     // even if Unknown, we proceed

                   yield new SourceTransactionDto(
                       bookedAt:         $bookedAt->startOfDay(),
                       postedAt:         $bookedAt,
                       valueDate:        $valueDate,
                       ownIban:          $ownIban,
                       counterpartyIban: ($r[AsnCsvColumnMap::COUNTERPARTY_IBAN] ?? '') !== '' ? (string) $r[AsnCsvColumnMap::COUNTERPARTY_IBAN] : null,
                       counterpartyName: trim((string) ($r[AsnCsvColumnMap::COUNTERPARTY_NAME] ?? '')) !== '' ? trim((string) $r[AsnCsvColumnMap::COUNTERPARTY_NAME]) : null,
                       currency:         (string) ($r[AsnCsvColumnMap::MUTATION_CURRENCY] ?? 'EUR'),
                       amountMinor:      $amountMinor,
                       sourceRef:        ($r[AsnCsvColumnMap::SEQUENCE_NUMBER] ?? '') !== '' ? (string) $r[AsnCsvColumnMap::SEQUENCE_NUMBER] : null,
                       description:      $this->joinDescription($r),
                       rawPayload:       $r,
                       sourceRowIndex:   $i,
                   );

                   $i++;
               }
           }

           private function joinDescription(array $row): ?string
           {
               $parts = [];
               foreach ([AsnCsvColumnMap::PAYMENT_REF, AsnCsvColumnMap::DESCRIPTION] as $col) {
                   $val = trim((string) ($row[$col] ?? ''));
                   if ($val !== '') $parts[] = $val;
               }
               if ($parts === []) return null;
               $combined = implode(' / ', $parts);
               // ASN descriptions can contain literal "\r" — replace with a space for storage:
               $combined = str_replace(["\r", "\n"], ' ', $combined);
               return preg_replace('/\s+/u', ' ', $combined);
           }
       }
       ```

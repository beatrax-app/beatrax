D-39 FX rate derivation formula (apply in NormalizeStage):

       <?php
       declare(strict_types=1);
       namespace Modules\Ingestion\Internal\Adapters\Ics;

       use Modules\Ingestion\Public\Exceptions\InvalidAmountException;

       /**
        * Parses ICS PDF amount strings into signed integer minor units.
        *
        * Supported shapes (drawn from Wave 0's ics-sample-1.md "Dutch amount
        * formats" section):
        *   - "€ 22,75"   → 2275
        *   - "-€ 22,75"  → -2275
        *   - "$ 12,99"   → 1299     (non-EUR currency leg of an FX row)
        *   - "12,99 USD" → 1299     (suffix-currency variant if present)
        *

       {
           public function parse(string $raw): int
           {
               $trimmed = trim($raw);
               if ($trimmed === '') {
                   throw new InvalidAmountException('Empty amount string.');
               }

               // Strip currency symbols (€, $, £, ¥) + ISO 4217 alpha codes
               // (USD, EUR, GBP, JPY, ...). Sign retained.
               $stripped = preg_replace('/[€$£¥]|\b[A-Z]{3}\b/u', '', $trimmed);
               if ($stripped === null) {
                   throw new InvalidAmountException(sprintf('Invalid amount string: %s', $raw));

               $stripped = trim($stripped);

               // Negative sign normalisation: accept prefix or suffix minus.
               $sign = 1;
               if (str_starts_with($stripped, '-')) {

                   $stripped = substr($stripped, 1);
               } elseif (str_ends_with($stripped, '-')) {
                   $sign = -1;
                   $stripped = substr($stripped, 0, -1);
               }
               $stripped = trim($stripped);

               // Dutch decimal: , is the decimal separator, . is thousands.
               // Collapse thousands separators, then split on comma.
               $stripped = str_replace('.', '', $stripped);
               $parts = explode(',', $stripped);
               if (count($parts) !== 2 || ! ctype_digit($parts[0]) || ! ctype_digit($parts[1]) || strlen($parts[1]) !== 2) {
                   throw new InvalidAmountException(sprintf('Invalid Dutch amount format: %s', $raw));
               }

               $minor = ((int) $parts[0]) * 100 + (int) $parts[1];
               return $sign * $minor;
           }
       }
